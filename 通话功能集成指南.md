### 2.5 添加 JS 通话逻辑

在 `<script>` 标签的最后（`</script>` 之前）添加：

```javascript
// ============================================================
// 语音通话功能
// ============================================================
let voiceCall = null;
let callTimerInterval = null;
let callStartTime = 0;
let callMuted = false;

// 从 CONFIG 推断语音后端 WebSocket URL
function getCallWsUrl() {
  // 如果后端部署在 Railway，语音后端可能是同一个域名的不同端口
  // 这里假设语音后端和主后端在同一个域名，端口 8004
  // 如果你的部署不同，请修改这里
  const base = CONFIG.HTTP_BASE || window.location.origin;
  try {
    const url = new URL(base);
    // 开发环境：localhost 直接用 8004 端口
    if (url.hostname === "localhost" || url.hostname === "127.0.0.1") {
      return `ws://${url.hostname}:8004/ws/call`;
    }
    // 生产环境：假设语音后端部署在同一个域名的 /voice 路径下（需要反向代理）
    // 或者直接用 wss://你的语音后端域名/ws/call
    return `wss://${url.host}/ws/call`;
  } catch {
    return "ws://localhost:8004/ws/call";
  }
}

async function startVoiceCall() {
  if (voiceCall && voiceCall.state !== "idle" && voiceCall.state !== "disconnected") {
    // 已经在通话中，点击则挂断
    hangupVoiceCall();
    return;
  }

  const conv = getCurrentConversation();
  if (!conv || conv.type !== "single") {
    showToast("通话功能仅支持单聊", "error");
    return;
  }

  const roleId = conv.roleId;
  const role = getRole(roleId);

  try {
    voiceCall = new VoiceCallModule({
      wsUrl: getCallWsUrl(),
      roleId: roleId,
      userId: currentUser ? currentUser.username : "guest",
      sessionId: currentUser ? user_db.get_session(currentUser.username) : null,

      onStateChange: (state) => {
        updateCallUIState(state, role);
      },
      onAsrPartial: (text) => {
        document.getElementById("call-user-text").textContent = text;
      },
      onAsrFinal: (text) => {
        document.getElementById("call-user-text").textContent = text;
        // 延迟清空用户字幕
        setTimeout(() => {
          if (document.getElementById("call-user-text").textContent === text) {
            document.getElementById("call-user-text").textContent = "";
          }
        }, 3000);
      },
      onAiReply: (text) => {
        document.getElementById("call-ai-text").textContent = text;
      },
      onVolume: (volume) => {
        const level = Math.min(100, volume * 800);
        document.getElementById("call-volume-level").style.width = level + "%";
      },
      onCallEnded: (transcript) => {
        handleCallEnded(transcript);
      },
      onError: (msg) => {
        showToast(msg, "error");
      },
    });

    const ok = await voiceCall.start();
    if (ok) {
      showCallOverlay(role);
      startCallTimer();
      document.getElementById("call-btn").classList.add("in-call");
    }
  } catch (err) {
    console.error("启动通话失败:", err);
    showToast("启动通话失败: " + (err.message || err), "error");
  }
}

function hangupVoiceCall() {
  if (voiceCall) {
    voiceCall.hangup();
  }
}

function showCallOverlay(role) {
  document.getElementById("call-overlay").classList.remove("hidden");
  document.getElementById("call-role-name").textContent = role.name;
  document.getElementById("call-role-avatar").textContent = role.emoji;
  document.getElementById("call-status-text").textContent = "正在连接...";
  document.getElementById("call-user-text").textContent = "";
  document.getElementById("call-ai-text").textContent = "";

  // 尝试在通话界面显示 Live2D（复用主界面的模型）
  // 简单方案：把主界面的 Live2D canvas 移动到通话区域
  // 这里用占位，实际可以根据需要实现
}

function hideCallOverlay() {
  document.getElementById("call-overlay").classList.add("hidden");
  document.getElementById("call-btn").classList.remove("in-call");
  stopCallTimer();
}

function updateCallUIState(state, role) {
  const statusEl = document.getElementById("call-status-text");
  const statusMap = {
    "connecting": "正在连接...",
    "listening": "聆听中...",
    "thinking": "思考中...",
    "speaking": "说话中...",
    "disconnected": "已断开",
  };
  statusEl.textContent = statusMap[state] || state;

  // 联动主界面的 Live2D 状态
  if (typeof setLive2DStatus === "function") {
    const live2dStatusMap = {
      "listening": ["listening", "聆听中"],
      "thinking": ["thinking", "思考中"],
      "speaking": ["speaking", "说话中"],
    };
    if (live2dStatusMap[state]) {
      setLive2DStatus(live2dStatusMap[state][0], live2dStatusMap[state][1]);
    }
  }
}

function startCallTimer() {
  callStartTime = Date.now();
  callTimerInterval = setInterval(() => {
    const elapsed = Math.floor((Date.now() - callStartTime) / 1000);
    const m = String(Math.floor(elapsed / 60)).padStart(2, "0");
    const s = String(elapsed % 60).padStart(2, "0");
    document.getElementById("call-timer").textContent = `${m}:${s}`;
  }, 1000);
}

function stopCallTimer() {
  if (callTimerInterval) {
    clearInterval(callTimerInterval);
    callTimerInterval = null;
  }
}

function handleCallEnded(transcript) {
  hideCallOverlay();

  // 把通话记录转成聊天消息
  if (transcript && transcript.length > 0) {
    const conv = getCurrentConversation();
    if (conv) {
      maybeAddTimeDivider(conv);
      transcript.forEach(item => {
        const msg = createMessage(item.role === "user" ? "user" : "ai", item.content);
        msg.isFromCall = true;
        conv.messages.push(msg);
        appendMessageToDOM(msg, conv);
      });
      renderConversationList();
      if (currentUser) saveConversationsToLocal(currentUser.username);
      scrollToBottom();
    }
  }

  // 更新 session_id
  if (voiceCall && voiceCall.session_id && currentUser) {
    user_db.set_session(currentUser.username, voiceCall.session_id);
  }

  showToast("通话已结束", "success");
}

// 通话按钮事件
document.getElementById("call-btn").addEventListener("click", startVoiceCall);

// 挂断按钮
document.getElementById("call-hangup-btn").addEventListener("click", hangupVoiceCall);

// 静音按钮
document.getElementById("call-mute-btn").addEventListener("click", () => {
  callMuted = !callMuted;
  const btn = document.getElementById("call-mute-btn");
  btn.classList.toggle("muted", callMuted);
  if (voiceCall && voiceCall.mediaStream) {
    voiceCall.mediaStream.getAudioTracks().forEach(t => {
      t.enabled = !callMuted;
    });
  }
  showToast(callMuted ? "已静音" : "已取消静音", callMuted ? "default" : "success");
});

// 扬声器按钮（占位，实际可以切换音频输出设备）
document.getElementById("call-speaker-btn").addEventListener("click", () => {
  showToast("扬声器切换功能开发中", "default");
});
```

## 第三步：main.py 修改

在 `main.py` 中，你**不需要大改**。通话功能是前端直接连语音后端的 WebSocket，不经过主后端。

但有两点需要确认：

### 3.1 CORS 配置

确保 `main.py` 的 FastAPI CORS 中间件允许 WebSocket 连接。如果你的前端和语音后端在不同域名/端口，需要在 `voice_server.py` 里配置 CORS（已经配置了 `allow_origins=["*"]`）。

### 3.2 反向代理（生产环境）

如果部署在 Railway 上，前端和语音后端可能是不同的服务。你需要：
- 要么给语音后端配一个独立域名（如 `voice.yourdomain.com`）
- 要么在主后端加一个反向代理，把 `/ws/call` 转发到语音后端的 8004 端口

如果需要反向代理，在 `main.py` 里加：

```python
from fastapi import WebSocket
from fastapi.middleware.wsgi import WSGIMiddleware
import httpx

# 反向代理 WebSocket 通话
@app.websocket("/ws/call")
async def proxy_call(websocket: WebSocket):
    await websocket.accept()
    async with httpx.AsyncClient() as client:
        # 连接语音后端
        async with client.websocket("ws://127.0.0.1:8004/ws/call") as upstream:
            # 双向转发
            async def forward_client():
                async for message in websocket.iter_messages():
                    if message["type"] == "websocket.receive":
                        if "bytes" in message:
                            await upstream.send_bytes(message["bytes"])
                        elif "text" in message:
                            await upstream.send_text(message["text"])
            async def forward_upstream():
                async for message in upstream.iter_bytes():
                    await websocket.send_bytes(message)
                async for message in upstream.iter_text():
                    await websocket.send_text(message)
            await asyncio.gather(forward_client(), forward_upstream())
```

> 注意：上面的反向代理代码是简化版，生产环境建议用 Nginx/Caddy 做反向代理，或者直接给语音后端配独立域名。

```bash
# 阿里云 NLS 实时 ASR
export ALIYUN_NLS_ACCESS_KEY_ID="你的阿里云AccessKeyId"

# 语音后端端口
export VOICE_PORT="8004"
```

3. 创建应用，获取 AppID 和 Access Token
4. 流式 TTS 需要开通"流式合成"权限

2. 进入单聊界面（念琦/清禾/璟雯）
3. 点击输入栏旁边的 📞 通话按钮

8. 点击挂断按钮，通话记录自动转成聊天消息

### 常见问题排查

| 问题 | 原因 | 解决 |
|---|---|---|
| 点击通话按钮没反应 | 浏览器不支持 AudioWorklet | 使用 Chrome/Edge 最新版 |
| 一直显示"正在连接" | WebSocket 连不上 | 检查语音后端是否启动、端口是否正确、防火墙 |
| 连接后立即断开 | 阿里云 NLS 配置错误 | 检查 AccessKeyId/Secret/AppKey 是否正确 |
| 能识别但没有语音回复 | 火山引擎 TTS 配置错误 | 检查 AppID/AccessToken/Cluster |
| 有语音但没声音 | 浏览器自动播放策略 | 通话界面已经有用户交互，应该可以播放；检查音量 |
| AI 说话时自己的声音被识别 | 回声抑制没生效 | 检查耳机是否插入（用耳机效果最好） |
| 通话记录没保存 | 记忆后端没启动 | 启动 vector_server.py，或忽略（不影响通话） |

## 架构说明

### 通话数据流
```
用户说话
  → 麦克风 (48kHz)
  → AudioWorklet 重采样 (16kHz PCM)
  → WebSocket 二进制帧 → 语音后端
  → 阿里云 NLS 实时 ASR
  → 识别结果 (JSON) → 前端显示字幕
  → 最终结果 → 人格后端 /api/generate
  → AI 回复文本 → 火山引擎流式 TTS
  → 音频分片 (PCM) → WebSocket 二进制帧 → 前端
  → AudioContext 流式播放
  → Live2D 嘴型同步
```

### 全双工实现
- **上行**：用户音频持续发送，不暂停
- **下行**：AI 音频分片到达即播放
- **打断**：VAD 检测到用户说话（音量超阈值持续 150ms）→ 自动发送 interrupt → 后端停止 TTS → 前端清空播放队列
- **回声抑制**：AI 说话时麦克风增益降到 10%，防止 AI 声音被自己的麦克风拾取

### 与现有功能的兼容
- 原有的"按住说话"语音消息功能**完全保留**，不受影响
- 通话结束后，通话记录自动转成普通聊天消息
- 通话和文字聊天共享同一个 session_id，上下文连续
- 群聊暂不支持通话（只支持单聊）
